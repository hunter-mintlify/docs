## 7. Inline expression

**Expected:** "2 + 2 = 4".

export const InlineCounter = () => {
  const [n, setN] = useState(0);
  return (
    <div className="not-prose flex items-center gap-3 rounded-xl border border-dashed border-zinc-300 dark:border-zinc-700 p-4 my-4">
      <span className="text-sm text-zinc-600 dark:text-zinc-400">Inline counter:</span>
      <span className="text-lg font-semibold">{n}</span>
      <button className="ml-auto rounded-lg bg-[#4D1DC4] px-3 py-1 text-sm text-white" onClick={() => setN(n + 1)}>
        +1
      </button>
    </div>
  );
};

export const InlineGreeting = ({ name }) => (
