title: "Docs trivia"
description: "A tiny quiz about writing documentation."
---

Test your documentation instincts. No pressure, there's no leaderboard.

<Trivia />

<Note>
Every answer here reflects common technical writing advice, not hard rules.
</Note>

export const Trivia = () => {
  const questions = [
    { q: "What should a page title describe?", options: ["The author's mood", "What the reader will learn or do", "The file name"], answer: 1 },
    { q: "When is a code example most helpful?", options: ["When it can be copied and run", "When it is a screenshot", "When it is very long"], answer: 0 },
    { q: "Where should the most important info go?", options: ["The footer", "A collapsed accordion", "Near the top of the page"], answer: 2 },
    { q: "What's the best length for a sentence in docs?", options: ["As short as it can be while staying clear", "At least 40 words", "Exactly one emoji"], answer: 0 }
  ];
  const [index, setIndex] = useState(0);
  const [picked, setPicked] = useState(null);
  const [score, setScore] = useState(0);
  const done = index >= questions.length;
  const choose = (i) => {
    if (picked !== null) return;
    setPicked(i);
    if (i === questions[index].answer) setScore(score + 1);
  };
  const next = () => { setPicked(null); setIndex(index + 1); };
  const reset = () => { setIndex(0); setPicked(null); setScore(0); };
  return (
    <div className="not-prose rounded-xl border border-zinc-200 dark:border-zinc-800 p-5 my-4">
      {done ? (
        <div>
          <div className="text-lg font-semibold">You scored {score} / {questions.length}</div>
          <div className="text-sm text-zinc-500 mt-1">{score === questions.length ? "Flawless. Go write some docs." : "Not bad. Try again?"}</div>
          <button onClick={reset} className="mt-4 rounded-md bg-[#16A34A] hover:bg-[#15803D] text-white px-4 py-1.5 text-sm font-medium">Play again</button>
        </div>
      ) : (
        <div>
          <div className="text-xs text-zinc-500 mb-2">Question {index + 1} of {questions.length}</div>
          <div className="font-semibold mb-3">{questions[index].q}</div>
          <div className="flex flex-col gap-2">
            {questions[index].options.map((opt, i) => {
              const isAnswer = i === questions[index].answer;
              let cls = "border-zinc-300 dark:border-zinc-700 hover:border-[#16A34A]";
              if (picked !== null && isAnswer) cls = "border-[#16A34A] bg-[#16A34A]/10";
              else if (picked === i) cls = "border-red-500 bg-red-500/10";
              return <button key={i} onClick={() => choose(i)} className={"text-left rounded-md border px-3 py-2 text-sm " + cls}>{opt}</button>;
            })}
          </div>
          {picked !== null && (
            <button onClick={next} className="mt-4 rounded-md bg-[#16A34A] hover:bg-[#15803D] text-white px-4 py-1.5 text-sm font-medium">Next</button>
          )}
        </div>
      )}
    </div>
  );
};
