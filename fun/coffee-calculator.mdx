Documentation runs on coffee. Work out the right ratio before your next writing sprint.

<CoffeeCalculator />

## Common ratios

| Style | Coffee to water |
| --- | --- |
| Strong | 1:14 |
| Balanced | 1:16 |
| Light | 1:18 |

<Warning>
This calculator does not know how many cups you've already had today.
</Warning>

export const CoffeeCalculator = () => {
  const [cups, setCups] = useState(2);
  const [ratio, setRatio] = useState(16);
  const water = cups * 250;
  const coffee = Math.round(water / ratio);
  return (
    <div className="not-prose rounded-xl border border-zinc-200 dark:border-zinc-800 p-5 my-4">
      <label className="block text-sm mb-4">
        <div className="flex justify-between mb-1"><span className="text-zinc-500">Cups (250 ml each)</span><span className="font-semibold">{cups}</span></div>
        <input type="range" min="1" max="8" value={cups} onChange={(e) => setCups(Number(e.target.value))} className="w-full accent-[#16A34A]" />
      </label>
      <label className="block text-sm mb-4">
        <div className="flex justify-between mb-1"><span className="text-zinc-500">Ratio</span><span className="font-semibold">1:{ratio}</span></div>
        <input type="range" min="12" max="20" value={ratio} onChange={(e) => setRatio(Number(e.target.value))} className="w-full accent-[#16A34A]" />
      </label>
      <div className="grid grid-cols-2 gap-3">
        <div className="rounded-lg bg-[#16A34A]/10 p-4">
          <div className="text-xs text-zinc-500">Coffee</div>
          <div className="text-2xl font-bold">{coffee} g</div>
        </div>
        <div className="rounded-lg bg-[#16A34A]/10 p-4">
          <div className="text-xs text-zinc-500">Water</div>
          <div className="text-2xl font-bold">{water} ml</div>
        </div>
      </div>
    </div>
  );
};
