export const sdBus = { l: {}, on(e, f) { (this.l[e] = this.l[e] || new Set()).add(f); return () => this.l[e].delete(f); }, emit(e, d) { (this.l[e] || []).forEach((f) => f(d)); } };

export const sdClamp = (v, a = 0, b = 1) => (v < a ? a : v > b ? b : v);

export const sdSmooth = (v) => v * v * (3 - 2 * v);

export const useSdSeen = (ref, threshold = 0.25) => {
  const [seen, setSeen] = useState(false);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const io = new IntersectionObserver((entries) => {
      if (entries.some((e) => e.isIntersecting)) {
        setSeen(true);
        io.disconnect();
      }
    }, { threshold });
    io.observe(el);
    return () => io.disconnect();
  }, []);
  return seen;
};

export const useSdVisible = (ref) => {
  const [visible, setVisible] = useState(false);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const io = new IntersectionObserver(([e]) => setVisible(e.isIntersecting), { threshold: 0.02 });
    io.observe(el);
    return () => io.disconnect();
  }, []);
  return visible;
};

export const useSdClock = (running, fps = 30) => {
  const [t, setT] = useState(0);
  const acc = useRef(0);
  useEffect(() => {
    if (!running) return;
    let raf = 0;
    let last = performance.now();
    let drawn = 0;
    const tick = (now) => {
      acc.current += Math.min(now - last, 100);
      last = now;
      if (now - drawn >= 1000 / fps) {
        drawn = now;
        setT(acc.current);
      }
      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [running]);
  const reset = () => {
    acc.current = 0;
    setT(0);
  };
  return [t, reset];
};

      const bendY = (z) => { const e = Math.max(0, -z) / (1.2 * LD.depth); return 160 * e * e; };

export const makeTorus = (canvas) => {
  const ctx = canvas.getContext('2d');
  const TAU = Math.PI * 2;
  const DEG = Math.PI / 180;
  const W = 640;
  const H = 524;
  const RM = 176;
  const RT = 66;
  const NU = 84;
  const NV = 42;
  const PERSP = 3600;
  const FRONT = [0, 57, 255];
  const BACK = [125, 125, 125];
  const SH_F = [6, 14, 70];
  const SH_B = [38, 38, 40];
  const LIGHT = [-0.5, -0.66, -0.56];
  const BAYER = [0, 32, 8, 40, 2, 34, 10, 42, 48, 16, 56, 24, 50, 18, 58, 26, 12, 44, 4, 36, 14, 46, 6, 38, 60, 28, 52, 20, 62, 30, 54, 22, 3, 35, 11, 43, 1, 33, 9, 41, 51, 19, 59, 27, 49, 17, 57, 25, 15, 47, 7, 39, 13, 45, 5, 37, 63, 31, 55, 23, 61, 29, 53, 21];
  const cosV = [];
  const sinV = [];
  for (let j = 0; j < NV; j++) { cosV.push(Math.cos((j * TAU) / NV)); sinV.push(Math.sin((j * TAU) / NV)); }
  const mod = (a) => { const r = a % TAU; return r < 0 ? r + TAU : r; };
  const rot = { cx: 1, sx: 0, cy: 1, sy: 0 };
  let rollC = 1;
  let rollS = 0;
  let scale = 1;
  let spin = 0;
  const tilt = { x: 0, y: 0, z: 0 };
  const want = { x: 0, y: 0, z: 0 };
  const pointer = { x: -1e5, y: -1e5 };
  const tmp = [0, 0, 0];
  const rotate = (x, y, z, out) => { const a = y * rot.cx - z * rot.sx; const b = y * rot.sx + z * rot.cx; out[0] = x * rot.cy + b * rot.sy; out[1] = a; out[2] = -x * rot.sy + b * rot.cy; return out; };
  const project = (x, y, z) => { rotate(x, y, z, tmp); const f = PERSP / (PERSP + tmp[2]); const px = tmp[0] * f; const py = tmp[1] * f; return [320 + px * rollC - py * rollS, 262 + px * rollS + py * rollC, tmp[2]]; };
  const smoother = (t, d) => { const x = sdClamp(t / d); return x + (x * x * x * (x * (6 * x - 15) + 10) - x) * 0.7; };
  const band = (time) => { const ph = time % 3.8; const head = smoother(ph, 2.4) * TAU; const tail = smoother(ph - 1.1, 2.4) * TAU; const len = Math.max(5 * DEG, head - tail); return [len, tail + len > TAU ? TAU - len : tail]; };
  const draw = (time, still) => {
    const wob = still ? 0 : 1;
    const ax = -64 * DEG + tilt.x + 0.03 * Math.sin(0.45 * time) * wob;
    const ay = tilt.y + 0.045 * Math.sin(0.3 * time) * wob;
    rot.cx = Math.cos(ax); rot.sx = Math.sin(ax); rot.cy = Math.cos(ay); rot.sy = Math.sin(ay);
    const roll = 28 * DEG + tilt.z;
    rollC = Math.cos(roll); rollS = Math.sin(roll);
    const [len0, tail] = still ? [60 * DEG, 0] : band(time);
    const start = mod(12 * DEG + spin + tail);
    const len = sdClamp(len0, 0.02, TAU - 1e-4);
    const rings = [];
    for (let i = 0; i < NU; i++) rings.push({ a: mod((i * TAU) / NU + spin), ring: i % 3 === 0 });
    rings.push({ a: start, ring: true });
    rings.push({ a: mod(start + len), ring: true });
    rings.sort((m, n) => m.a - n.a);
    let zmin = 1e9;
    let zmax = -1e9;
    const pts = rings.map(({ a }) => {
      const ca = Math.cos(a);
      const sa = Math.sin(a);
      const row = [];
      for (let j = 0; j < NV; j++) {
        const rr = RM + RT * cosV[j];
        const p = project(rr * ca, rr * sa, RT * sinV[j]);
        row.push(p);
        if (p[2] < zmin) zmin = p[2];
        if (p[2] > zmax) zmax = p[2];
      }
      return row;
    });
    const span = Math.max(1e-6, zmax - zmin);
    const quads = [];
    const nrm = [0, 0, 0];
    const pos = [0, 0, 0];
    const uStep = TAU / Math.max(8, Math.round((TAU * RM) / 3.6));
    const vStep = TAU / Math.max(8, Math.round((TAU * RT) / 3.6));
    for (let i = 0; i < rings.length; i++) {
      const k2 = (i + 1) % rings.length;
      const a0 = rings[i].a;
      let a1 = rings[k2].a;
      if (k2 === 0) a1 += TAU;
      if (a1 - a0 < 1e-5) continue;
      const mid = a0 + (a1 - a0) / 2;
      const inBand = mod(mid - start) <= len;
      const cm = Math.cos(mid);
      const sm = Math.sin(mid);
      for (let j = 0; j < NV; j++) {
        const line = j % 3 === 0;
        if (!inBand && !rings[i].ring && !line) continue;
        const jn = (j + 1) % NV;
        const A = pts[i][j];
        const B = pts[k2][j];
        const C = pts[k2][jn];
        const D = pts[i][jn];
        const q = { A, B, C, D, z: (A[2] + B[2] + C[2] + D[2]) / 4, ring: rings[i].ring, line, band: inBand, a0, a1, j, front: false, face: 0, lam: 0 };
        if (inBand) {
          const vm = ((j + 0.5) * TAU) / NV;
          const cv = Math.cos(vm);
          const sv = Math.sin(vm);
          rotate(cv * cm, cv * sm, sv, nrm);
          const rr = RM + RT * cv;
          rotate(rr * cm, rr * sm, RT * sv, pos);
          q.front = nrm[0] * pos[0] + nrm[1] * pos[1] + nrm[2] * (pos[2] + PERSP) < 0;
          q.face = Math.abs(nrm[2]);
          const lam = -(nrm[0] * LIGHT[0] + nrm[1] * LIGHT[1] + nrm[2] * LIGHT[2]) * (q.front ? 1 : -1);
          q.lam = lam > 0 ? lam : 0;
        }
        quads.push(q);
      }
    }
    quads.sort((m, n) => n.z - m.z);
    ctx.setTransform(scale, 0, 0, scale, 0, 0);
    ctx.clearRect(0, 0, W, H);
    ctx.lineJoin = 'round';
    ctx.lineCap = 'round';
    const bucket = (p, q) => Math.min(3, Math.floor((((p[2] + q[2]) / 2 - zmin) / span) * 4));
    const alphaOf = (b) => 1 - (0.3 * (b + 0.5)) / 4;
    let paths = [new Path2D(), new Path2D(), new Path2D(), new Path2D()];
    let used = [false, false, false, false];
    let pending = false;
    const flush = () => {
      if (!pending) return;
      ctx.strokeStyle = '#111111';
      ctx.lineWidth = 0.8;
      for (let b = 0; b < 4; b++) {
        if (used[b]) { ctx.globalAlpha = alphaOf(b); ctx.stroke(paths[b]); paths[b] = new Path2D(); used[b] = false; }
      }
      ctx.globalAlpha = 1;
      pending = false;
    };
    const batch = (p, q) => { const b = bucket(p, q); paths[b].moveTo(p[0], p[1]); paths[b].lineTo(q[0], q[1]); used[b] = true; pending = true; };
    const solo = (p, q) => { ctx.globalAlpha = alphaOf(bucket(p, q)); ctx.beginPath(); ctx.moveTo(p[0], p[1]); ctx.lineTo(q[0], q[1]); ctx.stroke(); };
    for (const q of quads) {
      if (!q.band) {
        if (q.ring) batch(q.A, q.D);
        if (q.line) batch(q.A, q.B);
        continue;
      }
      flush();
      const baseC = q.front ? FRONT : BACK;
      const shC = q.front ? SH_F : SH_B;
      const k = 0.12 * (1 - q.lam) * 0.85;
      const col = `rgb(${Math.round(baseC[0] + (shC[0] - baseC[0]) * k)},${Math.round(baseC[1] + (shC[1] - baseC[1]) * k)},${Math.round(baseC[2] + (shC[2] - baseC[2]) * k)})`;
      ctx.fillStyle = col;
      ctx.strokeStyle = col;
      ctx.lineWidth = 0.7;
      ctx.globalAlpha = 1;
      ctx.beginPath();
      ctx.moveTo(q.A[0], q.A[1]); ctx.lineTo(q.B[0], q.B[1]); ctx.lineTo(q.C[0], q.C[1]); ctx.lineTo(q.D[0], q.D[1]);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      if (q.front) {
        const r0 = q.a0 - spin;
        const r1 = q.a1 - spin;
        const iu0 = Math.ceil(r0 / uStep);
        const iu1 = Math.ceil(r1 / uStep) - 1;
        const v0 = (q.j * TAU) / NV;
        const v1 = ((q.j + 1) * TAU) / NV;
        const iv0 = Math.ceil(v0 / vStep);
        const iv1 = Math.ceil(v1 / vStep) - 1;
        if (iu1 >= iu0 && iv1 >= iv0) {
          const strength = 0.85 * (0.3 + 0.7 * q.face);
          let any = false;
          ctx.beginPath();
          const nU = Math.round(TAU / uStep);
          const nVv = Math.round(TAU / vStep);
          for (let iu = iu0; iu <= iu1; iu++) {
            const u = iu * uStep;
            const fu = (u + spin - q.a0) / (q.a1 - q.a0);
            const ex = q.A[0] + (q.B[0] - q.A[0]) * fu;
            const ey = q.A[1] + (q.B[1] - q.A[1]) * fu;
            const fx = q.D[0] + (q.C[0] - q.D[0]) * fu;
            const fy = q.D[1] + (q.C[1] - q.D[1]) * fu;
            const bu = ((iu % nU) + nU) % nU;
            for (let iv = iv0; iv <= iv1; iv++) {
              const v = iv * vStep;
              const fv = (v - v0) / (v1 - v0);
              const bv = ((iv % nVv) + nVv) % nVv;
              let w = 0.5 + 0.32 * Math.sin(2 * u + v + 1.7 * time) + 0.22 * Math.sin(4 * u - 2 * v - 1.15 * time + 2) + 0.12 * Math.sin(u + 4 * v + 0.8 * time + 4);
              w = (w - 0.5) * 2.2 + 0.5;
              if (w <= 0) continue;
              if (w > 1) w = 1;
              if ((BAYER[(bu % 8) * 8 + (bv % 8)] + 0.5) / 64 < w * strength) {
                ctx.rect(ex + (fx - ex) * fv - 0.75, ey + (fy - ey) * fv - 0.75, 1.5, 1.5);
                any = true;
              }
            }
          }
          if (any) { ctx.fillStyle = 'rgba(255,255,255,.92)'; ctx.fill(); }
        }
      }
      ctx.strokeStyle = '#111111';
      ctx.lineWidth = 0.8;
      if (q.ring) solo(q.A, q.D);
      if (q.line) solo(q.A, q.B);
      ctx.globalAlpha = 1;
    }
    flush();
  };
  const t0 = performance.now();
  let last = t0;
  return {
    resize() {
      const r = canvas.getBoundingClientRect().width / W || 1;
      scale = Math.min(3, (window.devicePixelRatio || 1) * r);
      canvas.width = Math.round(W * scale);
      canvas.height = Math.round(H * scale);
    },
    frame(now) {
      const dt = Math.min((now - last) / 1000, 0.1);
      last = now;
      const r = canvas.getBoundingClientRect();
      if (r.width) {
        const dx = pointer.x - (r.left + r.width / 2);
        const dy = pointer.y - (r.top + r.height / 2);
        const rad = 2.6 * Math.max(r.width, r.height);
        const d = Math.hypot(dx, dy);
        let o = Math.max(0, 1 - d / rad);
        o = sdSmooth(o);
        const nx = d > 1 ? dx / d : 0;
        const ny = d > 1 ? dy / d : 0;
        want.x = -ny * o * 0.18; want.y = nx * o * 0.18; want.z = nx * ny * o * 0.18 * 0.76;
      }
      const k = 1 - Math.exp(-6 * dt);
      tilt.x += (want.x - tilt.x) * k; tilt.y += (want.y - tilt.y) * k; tilt.z += (want.z - tilt.z) * k;
      spin += dt * 0.14;
      draw((now - t0) / 1000, false);
    },
    setPointer(x, y) { pointer.x = x; pointer.y = y; },
  };
};

export const SlashPage = () => {
  const [run, setRun] = useState(0);
  const [live, setLive] = useState(false);
  const [loading, setLoading] = useState(true);
  const rootRef = useRef(null);
  useEffect(() => {
    const root = rootRef.current;
    let raf = 0;
    let top0 = null;
    const onScroll = () => {
      if (raf) return;
      raf = requestAnimationFrame(() => {
        raf = 0;
        const top = root.getBoundingClientRect().top;
        if (top0 === null) top0 = top;
        sdBus.emit('sky:scroll', Math.max(0, top0 - top));
      });
    };
    document.addEventListener('scroll', onScroll, { passive: true, capture: true });
    onScroll();
    return () => { cancelAnimationFrame(raf); document.removeEventListener('scroll', onScroll, { capture: true }); };
  }, []);
  const replay = () => {
    const el = document.getElementById('sd-top');
    if (el) el.scrollIntoView({ block: 'start' });
    setLive(false);
    setLoading(true);
    setRun((n) => n + 1);
  };
  return (
    <div ref={rootRef} className={`sd sd-u not-prose ${live ? 'live' : ''}`}>
      <style>{SD_CSS}</style>
      {loading && <Loader key={run} onReveal={() => { setLive(true); sdBus.emit('loader:done'); }} onOff={() => setLoading(false)} />}
      <Nav live={live} />
      <div className="sd-light">
        <Sky key={`sky-${run}`} />
        <Storyboard key={`sb-${run}`} live={live} />
        <Fold />
      </div>
      <Values />
      <Stats />
      <Features />
      <Cta />
      <Faq />
      <Footer onReplay={replay} />
    </div>
  );
};
